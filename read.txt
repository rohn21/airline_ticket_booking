script usage
You can even allow choosing different settings modules like this:

# scripts/run.sh

#!/bin/bash
ENV=${1:-dev}
source /path/to/venv/bin/activate

if [ "$ENV" == "prod" ]; then
    export DJANGO_SETTINGS_MODULE=config.settings.prod
else
    export DJANGO_SETTINGS_MODULE=config.settings.dev
fi

python manage.py runserver

Usage:
./scripts/run.sh           # uses dev
./scripts/run.sh prod      # uses prod

# Run dev server
./scripts/run_dev.sh

# Run Gunicorn
./gunicorn/start.sh

# Run Celery
./scripts/run_celery.sh

# Apply migrations
./scripts/migrate.sh

# Run tests
./scripts/run_tests.sh
